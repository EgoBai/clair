根因证据 · 后端真实评分口径（clair-worker/worker.js buildSectorScores）
const breadthScore  = (upCount / count) * 100;                        // 上涨家数占比
const changeScore   = clamp(50 + avgChange * 6);                      // 带方向的涨跌幅变换
const volumeScore   = clamp(Math.log10(totalVolume / count + 1) * 8); // 成交量对数
const momentumScore = clamp(50 + avgChange * 5);
let score = momentum*0.35 + change*0.25 + breadth*0.25 + volume*0.15;  // 真实权重
